apiReference:
  title: Enviar entrada a una Session
  operation: Sessions
  method: POST
  path: /v1/agents/sessions/{session_id}/events
  description: 'Envía exactamente un evento por solicitud: agent.session.input.message o agent.session.input.cancel. Devuelve 202 sin cuerpo; sigue el progreso en el flujo de eventos. La forma de extensión de SandBase {"input": "…", "file_ids": [...]} envía texto con hasta 10 adjuntos listos y no se puede combinar con events. Un 409 session_input_pending significa que la entrada anterior todavía se está entregando: espera y no reenvíes con una clave nueva.'
  groups:
    - title: Parámetros de ruta
      fields:
        - name: session_id
          type: string
          required: true
          description: ID de la Session (prefijo ses_).
    - title: Encabezados
      fields:
        - name: Idempotency-Key
          type: string
          required: false
          description: Opcional, pero recomendado. Reutiliza la misma clave al reintentar la misma entrada.
    - title: Cuerpo de la solicitud
      description: events o bien input/file_ids.
      fields:
        - name: events
          type: array
          required: false
          description: Exactamente un evento. input.message lleva input como texto o como un mensaje de usuario input_text; input.cancel no lleva input.
        - name: input
          type: string
          required: false
          description: Extensión de SandBase. Texto no vacío de hasta 256 KiB.
        - name: file_ids
          type: array
          required: false
          description: Extensión de SandBase. Hasta 10 ID de adjuntos distintos con status ready.
