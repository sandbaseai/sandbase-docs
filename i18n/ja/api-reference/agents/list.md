apiReference:
  title: Agent の一覧
  operation: Agents
  method: GET
  path: /v1/agents
  description: 組織が所有する Agent を一覧表示します。結果はカーソルページネーションを使います。has_more が true の間は、last_id を after として渡してください。
  groups:
    - title: クエリパラメーター
      fields:
        - name: status
          type: string
          required: false
          description: active、archived、または all。その他の値は 400 を返します。
          default: active
        - name: limit
          type: integer
          required: false
          description: ページサイズ（1～100）。範囲外の値は 400 を返します。
          default: "20"
        - name: after
          type: string
          required: false
          description: カーソル。前のページの last_id を渡します。
        - name: order
          type: string
          required: false
          description: 作成日時の asc または desc。
          default: desc
